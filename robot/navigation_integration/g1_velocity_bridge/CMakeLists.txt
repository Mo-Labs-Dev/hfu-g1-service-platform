cmake_minimum_required(VERSION 3.16)
project(g1_velocity_bridge)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
set(UNITREE_SDK2_ROOT "$ENV{HOME}/unitree_sdk2" CACHE PATH "Existing Unitree SDK2 source")
set(SDK_DDS_LIB "${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
find_library(SDK_LIB unitree_sdk2 PATHS "${UNITREE_SDK2_ROOT}/lib/${CMAKE_SYSTEM_PROCESSOR}" NO_DEFAULT_PATH REQUIRED)
find_library(DDS_LIB ddsc PATHS "${SDK_DDS_LIB}" NO_DEFAULT_PATH REQUIRED)
find_library(DDSXX_LIB ddscxx PATHS "${SDK_DDS_LIB}" NO_DEFAULT_PATH REQUIRED)
add_executable(nav_velocity_bridge src/bridge.cpp)
add_executable(nav_waypoint_audio src/audio.cpp)
target_include_directories(nav_waypoint_audio PRIVATE include "${UNITREE_SDK2_ROOT}/include"
  "${UNITREE_SDK2_ROOT}/thirdparty/include" "${UNITREE_SDK2_ROOT}/thirdparty/include/ddscxx")
ament_target_dependencies(nav_waypoint_audio rclcpp std_msgs)
target_link_libraries(nav_waypoint_audio ${SDK_LIB} ${DDSXX_LIB} ${DDS_LIB} Threads::Threads nlohmann_json::nlohmann_json)
target_link_options(nav_waypoint_audio PRIVATE "-Wl,--disable-new-dtags")
set_target_properties(nav_waypoint_audio PROPERTIES INSTALL_RPATH "${SDK_DDS_LIB}")
target_include_directories(nav_velocity_bridge PRIVATE include "${UNITREE_SDK2_ROOT}/include"
  "${UNITREE_SDK2_ROOT}/thirdparty/include" "${UNITREE_SDK2_ROOT}/thirdparty/include/ddscxx")
ament_target_dependencies(nav_velocity_bridge rclcpp geometry_msgs nav_msgs std_msgs std_srvs tf2_ros tf2_geometry_msgs)
target_link_libraries(nav_velocity_bridge ${SDK_LIB} ${DDSXX_LIB} ${DDS_LIB} Threads::Threads)
# ROS Humble's LD_LIBRARY_PATH contains a different libddsc.so.0. Keep the
# SDK's ddscxx/ddsc pair together in this executable, without changing ROS.
target_link_options(nav_velocity_bridge PRIVATE "-Wl,--disable-new-dtags")
set_target_properties(nav_velocity_bridge PROPERTIES INSTALL_RPATH "${SDK_DDS_LIB}")
install(TARGETS nav_velocity_bridge nav_waypoint_audio DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_velocity_contract test/test_velocity_contract.cpp)
  target_include_directories(test_velocity_contract PRIVATE include)
  target_compile_options(test_velocity_contract PRIVATE -UNDEBUG)
  add_test(NAME velocity_contract COMMAND test_velocity_contract)
  add_executable(test_sdk_recovery test/test_sdk_recovery.cpp)
  target_include_directories(test_sdk_recovery PRIVATE include)
  target_compile_options(test_sdk_recovery PRIVATE -UNDEBUG)
  add_test(NAME sdk_recovery COMMAND test_sdk_recovery)
  # Same worker/callbacks as production, with a fake SDK that has no robot DDS.
  # Never installed; running it also requires an isolated localhost ROS domain.
  add_executable(nav_velocity_bridge_fault_test src/bridge.cpp)
  target_compile_definitions(nav_velocity_bridge_fault_test PRIVATE G1_BRIDGE_FAULT_TEST)
  target_include_directories(nav_velocity_bridge_fault_test PRIVATE include test)
  ament_target_dependencies(nav_velocity_bridge_fault_test rclcpp geometry_msgs nav_msgs
    std_msgs std_srvs tf2_ros tf2_geometry_msgs)
  target_link_libraries(nav_velocity_bridge_fault_test Threads::Threads)
  add_executable(test_audio_contract test/test_audio_contract.cpp)
  target_include_directories(test_audio_contract PRIVATE include)
  target_compile_options(test_audio_contract PRIVATE -UNDEBUG)
  add_test(NAME audio_contract COMMAND test_audio_contract)
endif()
ament_package()
